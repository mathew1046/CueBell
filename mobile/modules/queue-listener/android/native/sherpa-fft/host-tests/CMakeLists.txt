cmake_minimum_required(VERSION 3.22)
project(cuebell_gtcrn_regression CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_JNI OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_BINARY OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_BUILD_C_API_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_TTS OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_SPEAKER_DIARIZATION OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_WEBSOCKET OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_PORTAUDIO OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_LINK_LIBSTDCPP_STATICALLY OFF CACHE BOOL "" FORCE)
add_subdirectory("${CUEBELL_SHERPA_SOURCE}" sherpa)
add_executable(gtcrn-fft-regression ../test-gtcrn.cc)
target_include_directories(gtcrn-fft-regression PRIVATE "${CUEBELL_SHERPA_SOURCE}" "${CUEBELL_REFERENCE_HEADERS}")
target_link_libraries(gtcrn-fft-regression PRIVATE sherpa-onnx-core)
